from PIL import Image, ImageDraw def modifier_logo(logo_path, output_logo="logo_corec_final.png", output_html="telecharger_logo.html"): """ Script endepandan pou modifye logo COREC epi kreye yon bouton telechajman ki parèt anba a goch paj la. """ # Ouvri logo orijinal la logo = Image.open(logo_path).convert("RGBA") width, height = logo.size # Kreye yon kopi pou travay logo_no_orange = logo.copy() pixels = logo_no_orange.load() # Sant logo a cx, cy = width // 2, height // 2 rayon = min(width, height) // 4 + 20 # Retire cercle orange lan for x in range(width): for y in range(height): dx, dy = x - cx, y - cy if dx*dx + dy*dy <= rayon*rayon: r, g, b, a = pixels[x, y] if (r > 200 and g > 100 and b < 100): # koulè orange pixels[x, y] = (255, 255, 255, 0) # Ranplase cercle vèt la pa orange for x in range(width): for y in range(height): r, g, b, a = pixels[x, y] if g > 150 and r < 150 and b < 150: # koulè vèt pixels[x, y] = (255, 140, 0, a) # orange # Efase slogan an (zòn an ba logo a) draw = ImageDraw.Draw(logo_no_orange) draw.rectangle((cx - 200, cy + 80, cx + 200, cy + 160), fill=(255, 255, 255, 255)) # Sove nouvo logo a logo_no_orange.save(output_logo, "PNG") # Kreye yon paj HTML ak bouton telechajman fiks anba a goch html_code = f""" Telechaje Logo COREC

Logo COREC Final

""" with open(output_html, "w", encoding="utf-8") as f: f.write(html_code) print(f"✅ Nouvo logo anrejistre kòm: {output_logo}") print(f"✅ Paj telechajman kreye kòm: {output_html}")
